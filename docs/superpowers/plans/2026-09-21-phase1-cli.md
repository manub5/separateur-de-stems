# Séparateur de pistes — Plan d'implémentation Phase 1 (CLI)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construire un pipeline en ligne de commande qui sépare un fichier audio en pistes (voix, batterie, basse, guitare, piano, instrumental) en WAV 24 bits et MP3 320 kb/s, avec choix des pistes et sélection automatique des modèles.

**Tech Stack:** Python 3.12, `audio-separator` 0.47.0, `soundfile`, `pytest`, ffmpeg (présent sur le poste).

**Spec:** `docs/superpowers/specs/2026-09-21-separateur-de-stems-design.md`

```python
"""Output file naming helpers."""
import os
import re

_INVALID = re.compile(r'[<>:"/\\|?*]')


def sanitize(name: str) -> str:
    """Replace filesystem-invalid characters with underscores."""
    return _INVALID.sub("_", name)


def stem_filename(source_path: str, stem: str, ext: str, output_dir: str) -> str:
    """Build a unique output path: <source>_<stem>.<ext> in output_dir."""
    base = sanitize(os.path.splitext(os.path.basename(source_path))[0])
    stem = sanitize(stem)
    return unique_path(os.path.join(output_dir, f"{base}_{stem}.{ext}"))


def unique_path(path: str) -> str:
    """Return path unchanged if free, else append _1, _2, ... before extension."""
    if not os.path.exists(path):
        return path
    root, ext = os.path.splitext(path)
    i = 1
    while os.path.exists(f"{root}_{i}{ext}"):
        i += 1
    return f"{root}_{i}{ext}"
```

```python
"""Model catalog mapping and selection."""
from dataclasses import dataclass

from .errors import ModelUnavailableError

STEM_TO_MODEL: dict[str, str] = {
    "vocals": "vocals_mel_band_roformer.ckpt",
    "instrumental": "model_bs_roformer_ep_317_sdr_12.9755.ckpt",
    "drums": "htdemucs_ft.yaml",
    "bass": "hdemucs_mmi.yaml",
    "guitar": "htdemucs_6s.yaml",
    "piano": "htdemucs_6s.yaml",
}

SUPPORTED_EXTENSIONS: set[str] = {".wav", ".flac", ".mp3", ".aiff", ".aif", ".m4a"}

_MODEL_STEMS: dict[str, list[str]] = {
    "vocals_mel_band_roformer.ckpt": ["vocals", "instrumental"],
    "model_bs_roformer_ep_317_sdr_12.9755.ckpt": ["vocals", "instrumental"],
    "htdemucs_ft.yaml": ["vocals", "drums", "bass", "other"],
    "hdemucs_mmi.yaml": ["vocals", "drums", "bass", "other"],
    "htdemucs_6s.yaml": ["vocals", "drums", "bass", "guitar", "piano", "other"],
}


@dataclass(frozen=True)
class ModelSpec:
    filename: str
    stems: list[str]


def select_models(stems: set[str]) -> list[ModelSpec]:
    """Return the deduplicated list of models needed for the requested stems."""
    unknown = stems - set(STEM_TO_MODEL)
    if unknown:
        raise ModelUnavailableError(f"No model for stem(s): {', '.join(sorted(unknown))}")
    seen: list[str] = []
    for stem in sorted(stems):
        fn = STEM_TO_MODEL[stem]
        if fn not in seen:
            seen.append(fn)
    return [ModelSpec(filename=fn, stems=_MODEL_STEMS.get(fn, [])) for fn in seen]
```

```python
"""Access and verification of the real audio-separator model catalog."""
from .errors import ModelUnavailableError
from .models import STEM_TO_MODEL


def fetch_catalog(model_dir: str) -> dict:
    """Return the simplified model catalog {filename: {Name,Type,Stems,SDR}}."""
    from audio_separator.separator import Separator

    sep = Separator(info_only=True, model_file_dir=model_dir)
    return sep.get_simplified_model_list()


def verify_selected_models(catalog: dict, stems: set[str]) -> None:
    """Ensure every model required by stems is present in the catalog."""
    missing = [
        STEM_TO_MODEL[s] for s in sorted(stems) if STEM_TO_MODEL[s] not in catalog
    ]
    if missing:
        raise ModelUnavailableError(
            "Model(s) absent from catalog: " + ", ".join(sorted(set(missing)))
        )


def top_models_for_stem(catalog: dict, stem: str, limit: int = 5) -> list[tuple[str, float]]:
    """Return (filename, sdr) for models scoring stem, best first."""
    scored = [
        (fn, info["SDR"][stem])
        for fn, info in catalog.items()
        if info.get("SDR", {}).get(stem) is not None
    ]
    scored.sort(key=lambda x: x[1], reverse=True)
    return scored[:limit]
```

```python
"""Separation engine wrapping audio-separator (Qt-free)."""
import logging
import os
import threading

from .errors import (
    CancelledError, OutputError, StemSeparatorError, UnsupportedFormatError,
)
from .models import STEM_TO_MODEL, SUPPORTED_EXTENSIONS, select_models
from .naming import stem_filename


class SeparationEngine:
    def __init__(self, model_dir: str, output_dir: str, log_level: int = logging.INFO):
        self.model_dir = model_dir
        self.output_dir = output_dir
        self.log_level = log_level

    def _validate(self, input_path: str, cancel_event) -> None:
        if cancel_event is not None and cancel_event.is_set():
            raise CancelledError("Cancelled before start")
        if not os.path.isfile(input_path):
            raise UnsupportedFormatError(f"File not found: {input_path}")
        ext = os.path.splitext(input_path)[1].lower()
        if ext not in SUPPORTED_EXTENSIONS:
            raise UnsupportedFormatError(f"Unsupported format: {ext}")
        try:
            os.makedirs(self.output_dir, exist_ok=True)
        except OSError as exc:
            raise OutputError(f"Cannot write to output dir: {exc}") from exc

    def run(self, input_path, stems, progress_cb=None, cancel_event=None):
        from audio_separator.separator import Separator

        self._validate(input_path, cancel_event)
        specs = select_models(set(stems))
        results: dict[str, str] = {}
        total = len(specs)

        for index, spec in enumerate(specs):
            if cancel_event is not None and cancel_event.is_set():
                raise CancelledError("Cancelled")
            if progress_cb:
                progress_cb(int(index / total * 100), f"Loading {spec.filename}")
            try:
                separator = Separator(
                    model_file_dir=self.model_dir,
                    output_dir=self.output_dir,
                    output_format="WAV",
                    log_level=self.log_level,
                )
                separator.load_model(model_filename=spec.filename)
                if progress_cb:
                    progress_cb(int((index + 0.5) / total * 100), f"Separating {spec.filename}")
                outputs = separator.separate(input_path)
            except Exception as exc:
                raise OutputError(f"Separation failed for {spec.filename}: {exc}") from exc

            for out in outputs:
                stem = self._stem_from_output(out)
                if stem:
                    results[stem.lower()] = out
            if progress_cb:
                progress_cb(int((index + 1) / total * 100), f"Done {spec.filename}")

        return results

    @staticmethod
    def _stem_from_output(path: str) -> str | None:
        import re
        m = re.search(r"_\(([^)]+)\)", os.path.basename(path))
        return m.group(1) if m else None
```

```python
"""Output encoding: WAV 24-bit and MP3 320 kbps."""
import os
import subprocess

import soundfile as sf

from .errors import OutputError


def to_wav24(src: str, dest: str) -> str:
    """Rewrite src as a 24-bit PCM WAV file."""
    try:
        data, sr = sf.read(src, always_2d=True)
        sf.write(dest, data, sr, subtype="PCM_24")
    except Exception as exc:
        raise OutputError(f"WAV export failed: {exc}") from exc
    return dest


def to_mp3_320(src: str, dest: str) -> str:
    """Encode src to MP3 at 320 kbps using ffmpeg."""
    try:
        subprocess.run(
            ["ffmpeg", "-y", "-i", src, "-b:a", "320k", dest],
            check=True, capture_output=True,
        )
    except subprocess.CalledProcessError as exc:
        raise OutputError(f"MP3 export failed: {exc.stderr.decode(errors='replace')}") from exc
    except FileNotFoundError as exc:
        raise OutputError("ffmpeg not found in PATH") from exc
    return dest
```

```python
"""Command-line interface for stem separation."""
import argparse
import os
import sys

from .core.catalog import fetch_catalog, top_models_for_stem
from .core.engine import SeparationEngine
from .core.errors import StemSeparatorError, CancelledError
from .core.export import to_mp3_320, to_wav24
from .core.models import STEM_TO_MODEL
from .core.naming import stem_filename

DEFAULT_STEMS = "vocals,instrumental"


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="separateur-de-stems",
                                description="Separate an audio file into stems.")
    p.add_argument("input", nargs="?", help="Input audio file")
    p.add_argument("--stems", default=DEFAULT_STEMS,
                   help=f"Comma-separated stems (default: {DEFAULT_STEMS})")
    p.add_argument("--output-dir", default="output")
    p.add_argument("--model-dir", default="models")
    p.add_argument("--no-mp3", action="store_true")
    p.add_argument("--list-models", action="store_true")
    return p


def main(argv=None) -> int:
    args = build_parser().parse_args(argv)

    if args.list_models:
        catalog = fetch_catalog(args.model_dir)
        for stem in STEM_TO_MODEL:
            print(f"== {stem} ==")
            for fn, sdr in top_models_for_stem(catalog, stem, limit=5):
                print(f"  {fn}  SDR={sdr}")
        return 0

    if not args.input:
        print("error: input file required", file=sys.stderr)
        return 2

    stems = {s.strip().lower() for s in args.stems.split(",") if s.strip()}
    engine = SeparationEngine(model_dir=args.model_dir, output_dir=args.output_dir)

    def progress(pct, stage):
        print(f"[{pct:3d}%] {stage}", flush=True)

    try:
        results = engine.run(args.input, stems, progress_cb=progress)
    except CancelledError:
        print("Cancelled.", file=sys.stderr)
        return 130
    except StemSeparatorError as exc:
        print(f"error: {exc}", file=sys.stderr)
        return 2

    for stem, wav in results.items():
        wav24 = stem_filename(args.input, stem, "wav", args.output_dir)
        to_wav24(wav, wav24)
        print(f"WAV: {wav24}")
        if not args.no_mp3:
            mp3 = stem_filename(args.input, stem, "mp3", args.output_dir)
            to_mp3_320(wav24, mp3)
            print(f"MP3: {mp3}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Run separation in a killable subprocess for cancellable UI."""
import multiprocessing as mp

from .engine import SeparationEngine
from .errors import CancelledError, StemSeparatorError


def _worker(engine_kwargs, input_path, stems, queue):
    try:
        engine = SeparationEngine(**engine_kwargs)
        results = engine.run(input_path, stems)
        queue.put(("ok", results))
    except StemSeparatorError as exc:
        queue.put(("error", str(exc)))
    except Exception as exc:  # pragma: no cover - defensive
        queue.put(("error", f"unexpected: {exc}"))


class SubprocessSeparator:
    def __init__(self, engine_kwargs: dict):
        self.engine_kwargs = engine_kwargs
        self._cancelled = False
        self._process = None

    def cancel(self) -> None:
        self._cancelled = True
        if self._process is not None and self._process.is_alive():
            self._process.terminate()
            self._process.join()

    def run(self, input_path, stems):
        if self._cancelled:
            raise CancelledError("Cancelled")
        queue = mp.Queue()
        self._process = mp.Process(
            target=_worker, args=(self.engine_kwargs, input_path, stems, queue)
        )
        self._process.start()
        self._process.join()
        if self._cancelled:
            raise CancelledError("Cancelled")
        status, payload = queue.get()
        if status == "error":
            raise StemSeparatorError(payload)
        return payload
```

```python
"""Platform detection. macOS-specific behaviour is isolated and untested here."""
import platform


def is_apple_silicon() -> bool:
    return platform.system() == "Darwin" and platform.machine() == "arm64"


def torch_device_hint() -> str:
    """Best-effort device hint without importing torch."""
    if is_apple_silicon():
        return "mps"
    try:
        import torch
        return "cuda" if torch.cuda.is_available() else "cpu"
    except Exception:  # pragma: no cover - defensive
        return "cpu"
```
